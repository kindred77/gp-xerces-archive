    set(NETACCESSOR "socket")

  endif()

endif()

  if(WIN32)
    set(FILEMGR "windows")
  else()

# ---- install rules ----
install(TARGETS xerces-c
  EXPORT XercesCOTargets
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
  INCLUDES DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
)

install(DIRECTORY src/xercesc/
  DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/xercesc
  FILES_MATCHING
    PATTERN "*.hpp"
    PATTERN "*.h"
    PATTERN "*.c"
    PATTERN "Makefile*" EXCLUDE
)

install(FILES "${PROJECT_BINARY_DIR}/src/xercesc/util/Xerces_autoconf_config.hpp"
  DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/xercesc/util)

install(EXPORT XercesCOTargets
  FILE XercesCOTargets.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/xerces-c)

include(CMakePackageConfigHelpers)

  "${CMAKE_CURRENT_BINARY_DIR}/XercesCConfigVersion.cmake"

  COMPATIBILITY SameMajorVersion)

install(FILES "${CMAKE_CURRENT_BINARY_DIR}/XercesCConfigVersion.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/xerces-c)
